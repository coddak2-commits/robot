cmake_minimum_required(VERSION 3.16)
project(robot_camera LANGUAGES CXX)

# Windows(노트북) 빌드용. 리눅스(펜던트)는 build_camera_linux.sh 사용
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
add_definitions(-D_WIN32_WINNT=0x0A00 -DNOMINMAX -DWIN32_LEAN_AND_MEAN)

# Orbbec SDK: OrbbecSDK_v2.x_win_x64.zip 을 third_party/orbbec 에 풀어 둠 (git 에는 안 올림)
set(ORBBEC_ROOT "${CMAKE_SOURCE_DIR}/third_party/orbbec")
file(GLOB_RECURSE OB_HDR "${ORBBEC_ROOT}/*/libobsensor/ObSensor.hpp")
file(GLOB_RECURSE OB_LIB "${ORBBEC_ROOT}/*OrbbecSDK.lib")
file(GLOB_RECURSE OB_DLL "${ORBBEC_ROOT}/*OrbbecSDK.dll")
if(NOT OB_HDR OR NOT OB_LIB OR NOT OB_DLL)
    message(FATAL_ERROR "Orbbec SDK 없음: ${ORBBEC_ROOT} 에 OrbbecSDK_..._win_x64.zip 을 푸세요")
endif()
list(GET OB_HDR 0 OB_HDR)
list(GET OB_LIB 0 OB_LIB)
list(GET OB_DLL 0 OB_DLL)
get_filename_component(OB_INC "${OB_HDR}" DIRECTORY)
get_filename_component(OB_INC "${OB_INC}" DIRECTORY)
get_filename_component(OB_BIN "${OB_DLL}" DIRECTORY)
message(STATUS "Orbbec SDK: ${OB_INC} / ${OB_DLL}")

# MariaDB: robot-core 와 같은 방식 (vcpkg 우선, 없으면 로컬 설치)
find_package(unofficial-libmariadb CONFIG QUIET)
find_package(nlohmann_json CONFIG REQUIRED)

add_executable(robot_camera src/robot_camera.cpp)
target_include_directories(robot_camera PRIVATE "${CMAKE_SOURCE_DIR}/../robot-core/include" "${OB_INC}")
target_link_libraries(robot_camera PRIVATE "${OB_LIB}" nlohmann_json::nlohmann_json ws2_32)
if(unofficial-libmariadb_FOUND)
    target_link_libraries(robot_camera PRIVATE unofficial::libmariadb)
else()
    if(DEFINED ENV{MARIADB_DIR})
        set(MARIADB_DIR "$ENV{MARIADB_DIR}")
    else()
        set(MARIADB_DIR "C:/PROGRA~1/MARIAD~1.4")
    endif()
    target_include_directories(robot_camera PRIVATE "${MARIADB_DIR}/include/mysql")
    target_link_libraries(robot_camera PRIVATE "${MARIADB_DIR}/lib/libmariadb.lib")
endif()

# OrbbecSDK.dll + extensions 폴더를 실행 파일 옆에 복사
add_custom_command(TARGET robot_camera POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OB_DLL}" "$<TARGET_FILE_DIR:robot_camera>")
# SDK 가 같이 쓰는 VC 런타임 + 설정 파일 (있으면 복사)
foreach(F msvcp140_1.dll msvcp140_2.dll concrt140.dll OrbbecSDKConfig.xml)
    if(EXISTS "${OB_BIN}/${F}")
        add_custom_command(TARGET robot_camera POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${OB_BIN}/${F}" "$<TARGET_FILE_DIR:robot_camera>")
    endif()
endforeach()
if(EXISTS "${OB_BIN}/extensions")
    add_custom_command(TARGET robot_camera POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${OB_BIN}/extensions" "$<TARGET_FILE_DIR:robot_camera>/extensions")
endif()
